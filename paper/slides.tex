\begin{frame}{Step 2. AHP turns three competing criteria into one score}
  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      {\footnotesize
      \begin{tabular}{@{}ll@{}}
        \toprule
        \textbf{Criterion} & \textbf{Weight} \\
        \midrule
        Incident Density    & \textbf{\color{s1}67.4\%} \\
        Equity Gap          & 23.1\% \\
        Coverage Potential  & \phantom{0}9.4\% \\
        \bottomrule
      \end{tabular}}\\[1.0em]
      {\footnotesize\color{inks}Four team members rate the three criteria
      pairwise on Saaty's 1 to 9 scale, independently. The matrices are
      combined by \textbf{element-wise geometric mean}, the only average that
      keeps the matrix reciprocal.}
    \end{column}
    \begin{column}{0.5\textwidth}
      {\color{inks}Weights are the principal eigenvector. Pairwise human
      judgements can quietly contradict each other, so AHP measures that:}\\[0.6em]
      \centering
      $\mathrm{CI}=\dfrac{\lambda_{\max}-n}{n-1}, \qquad
       \mathrm{CR}=\dfrac{\mathrm{CI}}{\mathrm{RI}}$\\[0.9em]
      \begin{tabular}{@{}llll@{}}
        \toprule
        $\lambda_{\max}$ & CI & RI & CR \\
        \midrule
        3.0000 & 0.0000 & 0.58 & \textbf{\color{good}0.0000} \\
        \bottomrule
      \end{tabular}\\[0.7em]
      {\footnotesize\color{good}$\mathrm{CR}<0.10$ for all four members
      individually and for the aggregate.}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \caveat{CR is exactly zero by construction rather than by luck: each
  member's density-against-coverage rating is the product of their other two,
  so the judgements contain no contradiction for the check to find.}
\end{frame}

\begin{frame}{Step 3. Optimisation 1, where to put the stations}
  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\color{s1}\textbf{Mode A, set covering}}\\
      {\footnotesize\color{inks}Fewest stations that put every zone in reach.}\\[0.4em]
      \begin{align*}
        \min\ & \textstyle\sum_i y_i \\
        \text{s.t.}\ & \textstyle\sum_{i \in N(j)} y_i \ \geq\ 1 \quad \forall j \\
                     & y_i \in \{0,1\}
      \end{align*}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{s2}\textbf{Mode B, maximal covering}}\\
      {\footnotesize\color{inks}On a fixed budget $P$, cover the most demand.}\\[0.4em]
      \begin{align*}
        \max\ & \textstyle\sum_j d_j z_j \\
        \text{s.t.}\ & z_j \ \leq\ \textstyle\sum_{i \in N(j)} y_i \quad \forall j \\
                     & \textstyle\sum_i y_i = P, \quad y_i, z_j \in \{0,1\}
      \end{align*}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \hairline[0.5em]
  {\footnotesize\color{inks}$N(j)$ is the set of candidate sites that reach
  zone $j$ inside the limit. $d_j$ is the zone's incident count, so Mode B
  maximises people reached rather than area or zone count. Both are solved
  exactly with HiGHS via PuLP, in \textbf{5\,ms} on the study instance.}
\end{frame}

\begin{frame}{Step 4. Optimisation 2, who to send}
  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.44\textwidth}
      \begin{align*}
        \min\ & \textstyle\sum_i \sum_j c_{ij}\, x_{ij} \\
        \text{s.t.}\ & \textstyle\sum_i x_{ij} = 1 \quad \forall j \\
                     & \textstyle\sum_j x_{ij} \leq 1 \quad \forall i \\
                     & x_{ij} \geq 0
      \end{align*}
      {\footnotesize\color{inks}$c_{ij}$ is travel time from ambulance $i$ to
      incident $j$. The constraint matrix is totally unimodular, so the LP
      relaxation is already integral and no branching is needed.}
    \end{column}
    \begin{column}{0.52\textwidth}
      {\color{inks}\textbf{Scenario:} the busiest hour in the study year,
      09:00 on Wednesday 19 November 2025, with \textbf{30 simultaneous
      incidents}, dispatched from the \textbf{7 stations the covering model
      opened}.}\\[0.8em]
      {\footnotesize\color{inks}A rectangular cost matrix is padded with
      zero-cost dummy rows or columns so the problem is square and balanced.
      Dummy cells cost nothing, so they never distort the real total.}\\[0.8em]
      {\footnotesize\color{inkm}Cross-checked against an independent
      Hungarian-algorithm solution, which agrees to six decimal places.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{How the models were checked}
  \vspace{0.5em}
  {\color{inks}Each optimisation is validated against an independent method
  rather than assumed correct:}\\[0.9em]
  {\footnotesize\color{inks}
  $\bullet$\ \textbf{Set covering and maximal covering} were re-solved by
  \textbf{brute-force enumeration} on small instances, where every subset of
  candidate sites can be checked exhaustively. The ILP matched the enumerated
  optimum on each one.\\[0.6em]
  $\bullet$\ \textbf{AHP weights} were compared against the largest-modulus
  eigenvector computed directly by NumPy, over randomly generated reciprocal
  matrices.\\[0.6em]
  $\bullet$\ \textbf{The assignment LP} was cross-checked against an
  independent Hungarian-algorithm implementation, agreeing to six decimal
  places, and was confirmed never to lose to the greedy baseline over
  randomly generated cost matrices.\\[0.6em]
  $\bullet$\ \textbf{The robustness grid} is checked for monotonicity in both
  directions: a slower fleet or a twistier road network can never need fewer
  stations. That is what separates a real gradient from solver noise.}
\end{frame}

\begin{frame}{Conclusions}
  \vspace{0.6em}
  {\color{inks}
  $\bullet$\ Demand-driven placement reaches the whole city from
  \textbf{fewer sites} than Seattle's historically accumulated network. The
  finding is directional and survives every travel assumption we
  tested.\\[0.7em]
  $\bullet$\ The station count itself is \textbf{7 to 17}, not 7. Reporting
  the range is the result; reporting the point estimate would have been the
  mistake.\\[0.7em]
  $\bullet$\ Optimal dispatch beats nearest-unit-first by \textbf{8.8\%} of
  total fleet travel time on the busiest hour at the default fleet, measured
  like-for-like. The margin depends on fleet size, so it is only meaningful
  quoted with its configuration.\\[0.7em]
  $\bullet$\ The two models \textbf{compose}: covering output becomes
